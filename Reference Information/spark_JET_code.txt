### Output Cleaning 
 
import java.io.File 
import org.apache.commons.io.FileUtils 
 
val basePath = file_path + "Outputs/Parameter_Result/" 
 
val baseDir = new File(basePath) 
val exceptionFolders = baseDir.listFiles().filter(f => f.isDirectory && f.getName.startsWith("Parameter_Exception_")) 
 
exceptionFolders.foreach { folder => 
  val folderName = folder.getName 
  val exceptionNumber = folderName.replaceAll("[^0-9]", "") 
   
  try { 
    val csvFiles = folder.listFiles().filter(f => f.getName.endsWith(".csv") && !f.getName.endsWith(".csv.crc")) 
    if (csvFiles.nonEmpty) { 
      val latestCsv = csvFiles.maxBy(_.lastModified()) 
      val targetName = s"Parameter_Exception_${exceptionNumber}.csv" 
      val targetFile = new File(baseDir, targetName) 
       
      FileUtils.copyFile(latestCsv, targetFile) // copy instead of move to avoid permission issues 
      println(s"Copied ${latestCsv.getName} -> ${targetName}") 
    } 
 
    // Clean up files inside folder 
    folder.listFiles().foreach { f => 
      if (f.getName.endsWith(".crc") || f.getName == "_SUCCESS" || f.getName.endsWith(".csv.crc") || f.getName.endsWith(".csv")) { 
        try { 
          f.delete() 
          println(s"Deleted file: ${f.getName}") 
        } catch { 
          case e: Exception => println(s"Could not delete file ${f.getName}: ${e.getMessage}") 
        } 
      } 
    } 
 
    // Try to delete folder, but ignore if OneDrive locks it 
    try { 
      FileUtils.deleteDirectory(folder) 
      println(s"Cleaned folder: ${folderName}") 
      println(s"==============================================================================================================")   
    } catch { 
      case e: Exception => 
        println(s"Could not delete folder ${folderName}: ${e.getMessage}") 
        // Try to empty folder instead 
        folder.listFiles().foreach(_.delete()) 
    } 
  } catch { 
    case e: Exception => println(s"Skipping ${folderName}: ${e.getMessage}") 
  } 
} 
 
println("All CSVs processed. Locked folders were skipped safely.")
