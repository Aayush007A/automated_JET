        WHEN (Account_Subtype IN ('Assets', 'Asset')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Decrease in Assets'
        
        WHEN (Account_Subtype IN ('Liabilities', 'Liability')) AND (round(Amount_in_local_cur,2) > 0) THEN 'Decrease in Liabilities'
        WHEN (Account_Subtype IN ('Liabilities', 'Liability')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Increase in Liabilities'

        WHEN (Account_Subtype IN ('Expense', 'Expenses')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Decrease in Expense'
        
        WHEN (Account_Subtype IN ('Revenue', 'Revenues')) AND (round(Amount_in_local_cur,2) > 0) THEN 'Decrease in Revenue'
        WHEN (Account_Subtype IN ('Revenue', 'Revenues')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Increase in Revenue'

        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q3' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q3_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q4' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q4_Activity`

        WHEN (Account_Subtype = 'Equity') AND (round(Amount_in_local_cur,2) > 0) THEN 'Decrease in Equity'
        WHEN (Account_Subtype = 'Equity') AND (round(Amount_in_local_cur,2) < 0) THEN 'Increase in Equity'
        
        ELSE 'NA'
      END AS `Financial Statement Effect`,
      Quarter,
      Amount_in_local_cur
    FROM output_Exception6_1
    WHERE 
      (Account_Subtype IN ('Assets', 'Asset', 'Liabilities', 'Liability', 'Expense', 'Expenses', 'Revenue', 'Revenues', 'Equity', 'NA')) AND
      (Amount_in_local_cur != 0)
  )

  SELECT
    sa.Std_NonStd,
    sa.`Financial Statement Effect`,
    COUNT(*) AS `Total Nbr of Journal Entry Lines`,
    COALESCE(SUM(CASE WHEN sa.Amount_in_local_cur > 0 THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Total Debit Amount`,
        COALESCE(SUM(CASE WHEN sa.Amount_in_local_cur < 0 THEN -sa.Amount_in_local_cur ELSE 0 END), 0) AS `Total Credit Amount`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q1' THEN 1 ELSE 0 END), 0) AS `Q1_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q2' THEN 1 ELSE 0 END), 0) AS `Q2_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q3' THEN 1 ELSE 0 END), 0) AS `Q3_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q4' THEN 1 ELSE 0 END), 0) AS `Q4_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q1' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q1_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q2' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q2_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q3' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q3_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q4' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q4_Activity`

  FROM StandardAggregates sa
  GROUP BY sa.Std_NonStd, sa.`Financial Statement Effect`
  ORDER BY sa.Std_NonStd DESC, sa.`Financial Statement Effect`
""")


// COMMAND ----------

val ex11_summary_Stage1 = spark.sql("""
  WITH StandardAggregates AS (
    SELECT
      Std_NonStd,
      CASE
       WHEN (Account_Subtype IN ('Assets', 'Asset')) AND (round(Amount_in_local_cur,2) > 0) THEN 'Increase in Assets'
        WHEN (Account_Subtype IN ('Assets', 'Asset')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Decrease in Assets'
        WHEN (Account_Subtype IN ('Liabilities', 'Liability')) AND (round(Amount_in_local_cur,2) > 0) THEN 'Decrease in Liabilities'
        WHEN (Account_Subtype IN ('Liabilities', 'Liability')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Increase in Liabilities'
        WHEN (Account_Subtype IN ('Expense', 'Expenses')) AND (round(Amount_in_local_cur,2) > 0) THEN 'Increase in Expense'
        WHEN (Account_Subtype IN ('Expense', 'Expenses')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Decrease in Expense'
        WHEN (Account_Subtype IN ('Revenue', 'Revenues')) AND (round(Amount_in_local_cur,2) > 0) THEN 'Decrease in Revenue'
        WHEN (Account_Subtype IN ('Revenue', 'Revenues')) AND (round(Amount_in_local_cur,2) < 0) THEN 'Increase in Revenue'
        WHEN (Account_Subtype = 'Equity') AND (round(Amount_in_local_cur,2) > 0) THEN 'Decrease in Equity'
        WHEN (Account_Subtype = 'Equity') AND (round(Amount_in_local_cur,2) < 0) THEN 'Increase in Equity'
        ELSE 'NA'
      END AS `Financial Statement Effect`,
      Quarter,
      Amount_in_local_cur
    FROM output_Exception11
    WHERE 
      (Account_Subtype IN ('Assets', 'Asset', 'Liabilities', 'Liability', 'Expense', 'Expenses', 'Revenue', 'Revenues')) AND
      (Amount_in_local_cur != 0)
  )

  SELECT
    sa.Std_NonStd,
    sa.`Financial Statement Effect`,
    COUNT(*) AS `Total Nbr of Journal Entry Lines`,
    
        COALESCE(SUM(CASE WHEN sa.Amount_in_local_cur > 0 THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Total Debit Amount`,
        COALESCE(SUM(CASE WHEN sa.Amount_in_local_cur < 0 THEN -sa.Amount_in_local_cur ELSE 0 END), 0) AS `Total Credit Amount`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q1' THEN 1 ELSE 0 END), 0) AS `Q1_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q2' THEN 1 ELSE 0 END), 0) AS `Q2_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q3' THEN 1 ELSE 0 END), 0) AS `Q3_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q4' THEN 1 ELSE 0 END), 0) AS `Q4_Nbr of Journal Entries`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q1' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q1_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q2' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q2_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q3' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q3_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q4' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q4_Activity`

  FROM StandardAggregates sa
  GROUP BY sa.Std_NonStd, sa.`Financial Statement Effect`
  ORDER BY sa.Std_NonStd DESC, sa.`Financial Statement Effect`
""")

        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q1' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q1_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q2' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q2_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q3' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q3_Activity`,
        COALESCE(SUM(CASE WHEN sa.Quarter = 'Q4' THEN sa.Amount_in_local_cur ELSE 0 END), 0) AS `Q4_Activity`

  FROM StandardAggregates sa
  GROUP BY sa.Std_NonStd, sa.`Financial Statement Effect`

""")
